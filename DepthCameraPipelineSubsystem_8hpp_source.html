45 void addSample(const UpdateTimings& sample) {
46 m_sampleCount++;
47 const double count = static_cast<double>(m_sampleCount);
48 fusedOutputRetrievalSec += (sample.fusedOutputRetrievalSec - fusedOutputRetrievalSec) / count;
49 processUpdatesSec += (sample.processUpdatesSec - processUpdatesSec) / count;
50 totalSec += (sample.totalSec - totalSec) / count;
51 if (sample.pipelineExecuted) {
52 m_inputSampleCount++;
53 const double inputCount = static_cast<double>(m_inputSampleCount);
54 featureDetectionSec += (sample.featureDetectionSec - featureDetectionSec) / inputCount;
55 objectDetectionSec += (sample.objectDetectionSec - objectDetectionSec) / inputCount;
56 outputUpdateSec += (sample.outputUpdateSec - outputUpdateSec) / inputCount;
57 }
58 pipelineExecuted = sample.pipelineExecuted;
59 }
